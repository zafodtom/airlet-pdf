cmake_minimum_required(VERSION 3.24)
project(AirletPDF VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets PrintSupport)
if(WIN32 AND NOT Qt6_VERSION VERSION_EQUAL "6.8.3")
  message(FATAL_ERROR "Release builds require Qt 6.8.3; see docs/BUILD.md")
endif()
include(cmake/Pdfium.cmake)
add_library(airlet_core STATIC
  src/core/Geometry.cpp src/core/Geometry.h src/core/PdfEngine.h
  src/core/FolderNavigator.cpp src/core/FolderNavigator.h
  src/i18n/Language.cpp src/i18n/Language.h
  src/update/UpdateProtocol.cpp src/update/UpdateProtocol.h
  src/core/Selection.cpp src/core/Selection.h
  src/export/PageExport.cpp src/export/PageExport.h
  src/export/DocumentJobs.cpp src/export/DocumentJobs.h
  src/export/ImageSaveJob.cpp src/export/ImageSaveJob.h
  src/pdfium/PdfiumEngine.cpp src/pdfium/PdfiumEngine.h
  src/render/RenderService.cpp src/render/RenderService.h
  src/render/TileCache.h)
qt_add_resources(airlet_core airlet_translations PREFIX "/i18n" BASE src/i18n FILES src/i18n/en.json)
target_include_directories(airlet_core PUBLIC src)
target_link_libraries(airlet_core PUBLIC Qt6::Core Qt6::Gui Qt6::PrintSupport PRIVATE pdfium)
add_library(airlet_ui STATIC src/ui/PdfView.cpp src/ui/PdfView.h src/ui/MainWindow.cpp src/ui/MainWindow.h src/ui/DocumentTools.cpp src/ui/DocumentTools.h src/ui/UpdateController.cpp)
target_link_libraries(airlet_ui PUBLIC airlet_core Qt6::Widgets)
add_executable(AirletExport src/export/WorkerMain.cpp)
set_target_properties(AirletExport PROPERTIES WIN32_EXECUTABLE FALSE)
target_link_libraries(AirletExport PRIVATE airlet_core)
find_package(Qt6 REQUIRED COMPONENTS Network)
add_executable(AirletUpdate src/update/UpdateMain.cpp)
set_target_properties(AirletUpdate PROPERTIES WIN32_EXECUTABLE FALSE)
target_link_libraries(AirletUpdate PRIVATE airlet_core Qt6::Network Qt6::Widgets)
qt_add_resources(AirletUpdate update_icons PREFIX "/icons" BASE assets FILES assets/airlet.png)
if(WIN32)
  add_executable(AirletApplyUpdate WIN32 src/update/ApplyUpdate.cpp)
  set_target_properties(AirletApplyUpdate PROPERTIES AUTOMOC OFF)
  target_link_libraries(AirletApplyUpdate PRIVATE bcrypt shell32 gdi32)
  target_compile_definitions(AirletApplyUpdate PRIVATE UNICODE _UNICODE)
  if(MSVC)
    set_property(TARGET AirletApplyUpdate PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(AirletApplyUpdate PRIVATE /W4 /utf-8)
  endif()
  install(TARGETS AirletApplyUpdate RUNTIME DESTINATION .)
endif()
qt_add_executable(AirletPDF WIN32 src/main.cpp)
qt_add_resources(AirletPDF airlet_icons PREFIX "/icons" BASE assets FILES assets/airlet.png)
target_link_libraries(AirletPDF PRIVATE airlet_ui)
target_compile_definitions(AirletPDF PRIVATE AIRLET_VERSION="${PROJECT_VERSION}")
if(MSVC)
  foreach(t airlet_core airlet_ui AirletPDF AirletExport AirletUpdate)
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8)
  endforeach()
else()
  target_compile_options(airlet_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(airlet_ui PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  enable_language(RC)
  configure_file(resources/airlet.rc.in "${CMAKE_CURRENT_BINARY_DIR}/airlet.rc" @ONLY)
  target_sources(AirletPDF PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/airlet.rc")
  target_sources(AirletApplyUpdate PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/airlet.rc")
  # fpdfview.h includes windows.h. Its legacy min/max macros must not expand
  # std::min/std::max in the PDFium adapter.
  target_compile_definitions(airlet_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_custom_command(TARGET AirletPDF POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PDFIUM_ROOT}/bin/pdfium.dll" "$<TARGET_FILE_DIR:AirletPDF>")
  install(FILES "${PDFIUM_ROOT}/bin/pdfium.dll" DESTINATION .)
endif()
install(TARGETS AirletPDF AirletExport AirletUpdate RUNTIME DESTINATION .)
install(DIRECTORY licenses/ DESTINATION licenses)
install(FILES "${PDFIUM_ROOT}/LICENSE" DESTINATION licenses/pdfium)
install(DIRECTORY "${PDFIUM_ROOT}/licenses/" DESTINATION licenses/pdfium)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(airlet_tests tests/TestAirlet.cpp)
  qt_add_resources(airlet_tests airlet_test_icons PREFIX "/icons" BASE assets FILES assets/airlet.png)
  target_link_libraries(airlet_tests PRIVATE airlet_ui Qt6::Test)
  target_compile_definitions(airlet_tests PRIVATE EXPORT_EXE="$<TARGET_FILE:AirletExport>" FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  if(MSVC)
    target_compile_options(airlet_tests PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_dependencies(airlet_tests AirletPDF AirletExport AirletUpdate)
  if(WIN32)
    add_dependencies(airlet_tests AirletApplyUpdate)
    target_compile_definitions(airlet_tests PRIVATE APPLY_EXE="$<TARGET_FILE:AirletApplyUpdate>")
  endif()
  add_test(NAME airlet COMMAND airlet_tests -o "${CMAKE_CURRENT_BINARY_DIR}/airlet-tests.txt,txt")
  set_tests_properties(airlet PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(WIN32)
    add_custom_command(TARGET airlet_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PDFIUM_ROOT}/bin/pdfium.dll" "$<TARGET_FILE_DIR:airlet_tests>")
  endif()
endif()
